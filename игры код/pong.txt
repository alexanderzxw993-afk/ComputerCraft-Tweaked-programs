local w, h = term.getSize()
local PH = 4
local WIN = 5
local TICK = 0.05
local top = 2

-- сложности: скорость бота, скорость мяча, "реакция" бота
-- react: бот начинает следить за мячом, когда мяч прошёл эту долю ширины экрана
local levels = {
  {name = "EASY",   color = colors.lime,   botSpeed = 0.15, vstart = 0.35, vmax = 0.70, react = 0.55},
  {name = "NORMAL", color = colors.yellow, botSpeed = 0.30, vstart = 0.45, vmax = 0.95, react = 0.30},
  {name = "HARD",   color = colors.red,    botSpeed = 0.50, vstart = 0.60, vmax = 1.10, react = 0.00},
}
local D = levels[2]

local py, by
local bx, by_, vx, vy
local ps, bs = 0, 0

local function clamp(v, lo, hi)
  return math.max(lo, math.min(hi, v))
end

local function movePlayer(y)
  py = clamp(y, top, h - PH + 1)
end

local function serve(dir)
  bx, by_ = w / 2, (top + h) / 2
  vx = D.vstart * dir
  vy = (math.random() - 0.5) * 0.5
end

local function draw()
  term.setBackgroundColor(colors.black)
  term.clear()
  term.setTextColor(colors.yellow)
  term.setCursorPos(1, 1)
  write("YOU " .. ps)
  local mid = "[" .. D.name .. "]"
  term.setTextColor(D.color)
  term.setCursorPos(math.floor((w - #mid) / 2) + 1, 1)
  write(mid)
  term.setTextColor(colors.yellow)
  local s = "BOT " .. bs
  term.setCursorPos(w - #s + 1, 1)
  write(s)

  term.setBackgroundColor(colors.white)
  for i = 0, PH - 1 do
    term.setCursorPos(2, py + i)
    write(" ")
    term.setCursorPos(w - 1, math.floor(by + 0.5) + i)
    write(" ")
  end

  term.setBackgroundColor(colors.yellow)
  term.setCursorPos(clamp(math.floor(bx + 0.5), 1, w), clamp(math.floor(by_ + 0.5), top, h))
  write(" ")
  term.setBackgroundColor(colors.black)
end

local function bounceOff(padY, dir)
  local hit = (by_ - (padY + PH / 2 - 0.5)) / (PH / 2)
  vx = clamp(math.abs(vx) * 1.06, D.vstart, D.vmax) * dir
  vy = clamp(hit * 0.6, -0.7, 0.7)
end

local function round()
  ps, bs = 0, 0
  py = math.floor((top + h - PH) / 2)
  by = py
  serve(math.random(2) == 1 and 1 or -1)
  draw()
  sleep(1)

  local timer = os.startTimer(TICK)
  while ps < WIN and bs < WIN do
    local e, a, b, c = os.pullEvent()
    if e == "key" then
      if a == keys.up then movePlayer(py - 1)
      elseif a == keys.down then movePlayer(py + 1)
      elseif a == keys.q then return false end
    elseif e == "mouse_click" or e == "mouse_drag" then
      movePlayer(c - math.floor(PH / 2))
    elseif e == "timer" and a == timer then
      bx, by_ = bx + vx, by_ + vy
      if by_ < top then by_ = top; vy = -vy end
      if by_ > h then by_ = h; vy = -vy end

      if vx < 0 and bx <= 3 then
        if by_ >= py - 0.5 and by_ <= py + PH - 0.5 then
          bx = 3
          bounceOff(py, 1)
        elseif bx < 1 then
          bs = bs + 1
          serve(1)
          draw()
          sleep(0.8)
        end
      end

      if vx > 0 and bx >= w - 2 then
        if by_ >= by - 0.5 and by_ <= by + PH - 0.5 then
          bx = w - 2
          bounceOff(by, -1)
        elseif bx > w then
          ps = ps + 1
          serve(-1)
          draw()
          sleep(0.8)
        end
      end

      -- бот: следит за мячом, если мяч летит к нему и прошёл линию реакции
      local center = (top + h - PH) / 2
      local target = center
      if vx > 0 and bx > w * D.react then
        target = by_ - PH / 2 + 0.5
      end
      local diff = target - by
      by = clamp(by + clamp(diff, -D.botSpeed, D.botSpeed), top, h - PH + 1)

      draw()
      timer = os.startTimer(TICK)
    end
  end
  return true
end

-- меню выбора сложности
local function menu()
  local sel = 2
  while true do
    term.setBackgroundColor(colors.black)
    term.clear()
    term.setTextColor(colors.yellow)
    term.setCursorPos(2, 2)
    write("PONG vs BOT")
    term.setTextColor(colors.white)
    term.setCursorPos(2, 4)
    write("Choose difficulty:")
    for i, l in ipairs(levels) do
      term.setCursorPos(4, 5 + i)
      term.setTextColor(i == sel and l.color or colors.gray)
      write((i == sel and "> " or "  ") .. i .. ". " .. l.name)
    end
    term.setTextColor(colors.lightGray)
    term.setCursorPos(2, 11)
    write("1-3 / arrows + ENTER / mouse")
    term.setCursorPos(2, 12)
    write("Q - quit")

    local e, a, b, c = os.pullEvent()
    if e == "key" then
      if a == keys.up then sel = (sel + 1) % 3 + 1
      elseif a == keys.down then sel = sel % 3 + 1
      elseif a == keys.one then D = levels[1] return true
      elseif a == keys.two then D = levels[2] return true
      elseif a == keys.three then D = levels[3] return true
      elseif a == keys.enter then D = levels[sel] return true
      elseif a == keys.q then return false end
    elseif e == "mouse_click" then
      local i = c - 5
      if i >= 1 and i <= 3 then D = levels[i] return true end
    end
  end
end

math.randomseed(os.epoch("utc"))

local alive = menu()
while alive do
  if not round() then break end
  term.setBackgroundColor(colors.black)
  term.clear()
  term.setTextColor(ps >= WIN and colors.lime or colors.red)
  term.setCursorPos(2, 3)
  write(ps >= WIN and "YOU WIN!" or "BOT WINS")
  term.setTextColor(colors.white)
  term.setCursorPos(2, 5)
  write("Score: " .. ps .. " : " .. bs .. "  (" .. D.name .. ")")
  term.setCursorPos(2, 7)
  write("ENTER - again, M - menu, Q - quit")
  while true do
    local _, k = os.pullEvent("key")
    if k == keys.enter then break end
    if k == keys.m then alive = menu() break end
    if k == keys.q then alive = false break end
  end
end

term.setBackgroundColor(colors.black)
term.setTextColor(colors.white)
term.clear()
term.setCursorPos(1, 1)
