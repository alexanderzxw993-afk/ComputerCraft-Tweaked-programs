local w, h = term.getSize()
local DIR = "mynotes"
if not fs.exists(DIR) then fs.makeDir(DIR) end

local IMP_FILE = fs.combine(DIR, ".important")
local important = {}
local items, sel, top = {}, 1, 1

local function loadImportant()
  important = {}
  if fs.exists(IMP_FILE) then
    local f = fs.open(IMP_FILE, "r")
    local text = f.readAll() or ""
    f.close()
    for line in (text .. "\n"):gmatch("(.-)\n") do
      if line ~= "" then important[line] = true end
    end
  end
end

local function saveImportant()
  local f = fs.open(IMP_FILE, "w")
  for name in pairs(important) do f.writeLine(name) end
  f.close()
end

local function firstLine(path)
  local f = fs.open(path, "r")
  if not f then return "" end
  local line = f.readLine() or ""
  f.close()
  return line
end

local function scan()
  local list = {}
  for _, f in ipairs(fs.list(DIR)) do
    local path = fs.combine(DIR, f)
    if not fs.isDir(path) and f:match("%.txt$") then
      local base = f:gsub("%.txt$", "")
      list[#list + 1] = {
        name = base,
        path = path,
        preview = firstLine(path),
      }
    end
  end
  table.sort(list, function(a, b)
    local ia, ib = important[a.name] and 1 or 0, important[b.name] and 1 or 0
    if ia ~= ib then return ia > ib end
    return a.name:lower() < b.name:lower()
  end)
  return list
end

local function refresh(keepName)
  items = scan()
  sel = 1
  if keepName then
    for i, it in ipairs(items) do
      if it.name == keepName then sel = i break end
    end
  end
end

local function status(text, color)
  term.setBackgroundColor(colors.black)
  term.setTextColor(color or colors.red)
  term.setCursorPos(1, h - 1)
  term.clearLine()
  write(text)
end

local function draw()
  term.setBackgroundColor(colors.black)
  term.clear()
  term.setTextColor(colors.yellow)
  term.setCursorPos(1, 1)
  write("NOTES  (" .. #items .. ")")

  local rows = h - 4
  if sel < top then top = sel end
  if sel > top + rows - 1 then top = sel - rows + 1 end
  if top < 1 then top = 1 end

  if #items == 0 then
    term.setTextColor(colors.lightGray)
    term.setCursorPos(2, 3)
    write("No notes yet. Press N to create one.")
  end

  for i = 0, rows - 1 do
    local it = items[top + i]
    if it then
      local selected = (top + i == sel)
      local imp = important[it.name]
      term.setCursorPos(1, 3 + i)
      term.setBackgroundColor(selected and colors.white or colors.black)
      term.clearLine()
      if selected then
        term.setTextColor(imp and colors.red or colors.black)
      else
        term.setTextColor(imp and colors.yellow or colors.white)
      end
      write((imp and "*" or " ") .. it.name)
      local used = #it.name + 3
      local room = w - used - 1
      if room > 4 and it.preview ~= "" then
        term.setTextColor(selected and colors.gray or colors.lightGray)
        write(" " .. it.preview:sub(1, room - 1))
      end
    end
  end

  term.setBackgroundColor(colors.black)
  term.setTextColor(colors.gray)
  term.setCursorPos(1, h)
  write("ENTER open N new I star D del R ren Q quit")
end

local function cleanName(s)
  s = s:gsub("^%s+", "")
  s = s:gsub("%s+$", "")
  s = s:gsub("[^%w _%-]", "_")
  return s
end

local function openNote(it)
  term.setBackgroundColor(colors.black)
  term.setTextColor(colors.white)
  term.clear()
  term.setCursorPos(1, 1)
  shell.run("edit", it.path)
  refresh(it.name)
end

local function newNote()
  status("Note name: ", colors.yellow)
  local name = cleanName(read())
  if name == "" then return end
  local path = fs.combine(DIR, name .. ".txt")
  if not fs.exists(path) then
    local f = fs.open(path, "w")
    f.write("")
    f.close()
  end
  refresh(name)
  if items[sel] then openNote(items[sel]) end
end

local function deleteNote(it)
  status("Delete " .. it.name .. "? Y/N")
  while true do
    local _, k = os.pullEvent("key")
    if k == keys.y then
      fs.delete(it.path)
      important[it.name] = nil
      saveImportant()
      refresh()
      return
    elseif k == keys.n then
      return
    end
  end
end

local function renameNote(it)
  status("New name: ", colors.yellow)
  local name = cleanName(read())
  if name == "" then return end
  local newPath = fs.combine(DIR, name .. ".txt")
  if fs.exists(newPath) then
    status("Name already exists")
    sleep(1.5)
  else
    fs.move(it.path, newPath)
    if important[it.name] then
      important[it.name] = nil
      important[name] = true
      saveImportant()
    end
    refresh(name)
  end
end

local function toggleImportant(it)
  if important[it.name] then
    important[it.name] = nil
  else
    important[it.name] = true
  end
  saveImportant()
  refresh(it.name)
end

loadImportant()
refresh()
while true do
  draw()
  local e, a, b, c = os.pullEvent()
  if e == "key" then
    if a == keys.up and sel > 1 then sel = sel - 1
    elseif a == keys.down and sel < #items then sel = sel + 1
    elseif a == keys.enter and items[sel] then openNote(items[sel])
    elseif a == keys.n then newNote()
    elseif a == keys.i and items[sel] then toggleImportant(items[sel])
    elseif a == keys.d and items[sel] then deleteNote(items[sel])
    elseif a == keys.r and items[sel] then renameNote(items[sel])
    elseif a == keys.q then break end
  elseif e == "mouse_click" then
    local idx = top + (c - 3)
    if c >= 3 and items[idx] then
      sel = idx
      openNote(items[idx])
    end
  elseif e == "mouse_scroll" then
    sel = math.max(1, math.min(math.max(1, #items), sel + a))
  end
end

term.setBackgroundColor(colors.black)
term.setTextColor(colors.white)
term.clear()
term.setCursorPos(1, 1)
