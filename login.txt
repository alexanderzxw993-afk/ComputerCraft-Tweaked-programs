print("Starting...")

local IDLE = 60               -- секунд бездействия до скринсейвера
local SWITCH_MS = 20000       -- смена фона, мс
local LOCK_AFTER_SAVER = true -- спрашивать пароль после скринсейвера
local DATA = "login.dat"

local tz = 3                  -- МСК, либо из clock.cfg
if fs.exists("clock.cfg") then
  local f = fs.open("clock.cfg", "r")
  local t = textutils.unserialize(f.readAll() or "")
  f.close()
  if type(t) == "table" and tonumber(t.offset) then tz = tonumber(t.offset) end
end

local WDAY = {"Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"}

local function hash(pw, salt)
  local s = salt .. pw .. salt
  local h1, h2 = 12345, 67890
  for round = 1, 500 do
    for i = 1, #s do
      local b = s:byte(i)
      h1 = (h1 * 48271 + b + round) % 2147483647
      h2 = (h2 * 69621 + b * 7 + h1) % 2147483647
    end
  end
  return string.format("%08x%08x", h1, h2)
end

local function loadRec()
  if not fs.exists(DATA) then return nil end
  local f = fs.open(DATA, "r")
  local t = textutils.unserialize(f.readAll() or "")
  f.close()
  if type(t) == "table" and t.salt and t.hash then return t end
end

local function saveRec(pw)
  local salt = tostring(os.epoch("utc")) .. tostring(math.random(100000, 999999))
  local f = fs.open(DATA, "w")
  f.write(textutils.serialize({salt = salt, hash = hash(pw, salt)}))
  f.close()
end

local FONT = {
  ["0"] = {"111", "101", "101", "101", "111"},
  ["1"] = {"010", "110", "010", "010", "111"},
  ["2"] = {"111", "001", "111", "100", "111"},
  ["3"] = {"111", "001", "111", "001", "111"},
  ["4"] = {"101", "101", "111", "001", "001"},
  ["5"] = {"111", "100", "111", "001", "111"},
  ["6"] = {"111", "100", "111", "101", "111"},
  ["7"] = {"111", "001", "001", "001", "001"},
  ["8"] = {"111", "101", "111", "101", "111"},
  ["9"] = {"111", "101", "111", "001", "111"},
}

local CH = "0123456789ABCDEFXYZ#$%&*+<>=?"
local function rc()
  local i = math.random(#CH)
  return CH:sub(i, i)
end

----------------------------------------------------------------
-- сцена: один экран (компьютер или монитор)
----------------------------------------------------------------
local function makeScene(out)
  local W, H = out.getSize()
  local S = {out = out}
  local txt, fg, bg = {}, {}, {}

  local function alloc()
    W, H = out.getSize()
    txt, fg, bg = {}, {}, {}
    for y = 1, H do txt[y], fg[y], bg[y] = {}, {}, {} end
  end

  local function blank()
    for y = 1, H do
      local t, f, b = txt[y], fg[y], bg[y]
      for x = 1, W do t[x] = " "; f[x] = "0"; b[x] = "f" end
    end
  end

  local function setc(x, y, c)
    if x >= 1 and x <= W and y >= 1 and y <= H then bg[y][x] = c end
  end

  local function text(y, str, color)
    if not str or str == "" or y < 1 or y > H then return end
    str = " " .. str .. " "
    local x0 = math.floor((W - #str) / 2) + 1
    for i = 1, #str do
      local x = x0 + i - 1
      if x >= 1 and x <= W then
        txt[y][x] = str:sub(i, i)
        fg[y][x] = color
        bg[y][x] = "f"
      end
    end
  end

  local modes = {}

  -- ОГОНЬ
  local heat
  local FIRE = "fce140"
  modes.fire = {
    init = function()
      heat = {}
      for y = 1, H do
        heat[y] = {}
        for x = 1, W do heat[y][x] = 0 end
      end
    end,
    draw = function()
      for x = 1, W do heat[H][x] = math.random(18, 35) end
      local cool = 72 / H
      for y = 1, H - 1 do
        local b1, b2, row = heat[y + 1], heat[math.min(H, y + 2)], heat[y]
        for x = 1, W do
          local v = (b1[math.max(1, x - 1)] + b1[x] + b1[math.min(W, x + 1)] + b2[x]) / 4
                    - math.random() * cool
          row[x] = v > 0 and v or 0
        end
      end
      for y = 1, H do
        for x = 1, W do
          local i = math.floor(heat[y][x] / 6) + 1
          if i > 6 then i = 6 end
          bg[y][x] = FIRE:sub(i, i)
        end
      end
    end,
  }

  -- МАТРИЦА
  local inten, chr, heads
  modes.matrix = {
    init = function()
      inten, chr, heads = {}, {}, {}
      for y = 1, H do
        inten[y], chr[y] = {}, {}
        for x = 1, W do inten[y][x] = 0; chr[y][x] = " " end
      end
      for x = 1, W do heads[x] = {y = -math.random(H), spd = math.random(1, 3)} end
    end,
    draw = function(f)
      for x = 1, W do
        local h = heads[x]
        if f % h.spd == 0 then
          h.y = h.y + 1
          if h.y >= 1 and h.y <= H then
            inten[h.y][x] = 1
            chr[h.y][x] = rc()
          end
          if h.y > H + 6 then
            h.y = -math.random(H)
            h.spd = math.random(1, 3)
          end
        end
      end
      for y = 1, H do
        for x = 1, W do
          local v = inten[y][x]
          if v > 0.02 then
            txt[y][x] = chr[y][x]
            fg[y][x] = v > 0.93 and "0" or v > 0.5 and "5" or v > 0.2 and "d" or "7"
            inten[y][x] = v - 0.045
          end
        end
      end
    end,
  }

  -- ЗВЁЗДЫ (крупные, с хвостами)
  local stars
  local SC = {"0", "0", "0", "9", "3", "4", "6"}
  local function newStar(s)
    s.x = (math.random() - 0.5) * 2
    s.y = (math.random() - 0.5) * 2
    s.z = math.random() * 1.0 + 0.3
    s.c = SC[math.random(#SC)]
  end
  local function proj(s, z)
    return math.floor(W / 2 + s.x / z * W / 2 + 0.5),
           math.floor(H / 2 + s.y / z * H / 2 + 0.5)
  end
  modes.stars = {
    init = function()
      stars = {}
      for i = 1, math.max(25, math.floor(W * H / 14)) do
        stars[i] = {}
        newStar(stars[i])
      end
    end,
    draw = function()
      for _, s in ipairs(stars) do
        s.z = s.z - 0.025
        local sx, sy = proj(s, math.max(s.z, 0.05))
        if s.z <= 0.1 or sx < 1 or sx > W or sy < 1 or sy > H then
          newStar(s)
        else
          local px, py = proj(s, s.z + 0.07)
          local steps = math.max(math.abs(sx - px), math.abs(sy - py), 1)
          for i = 0, steps - 1 do
            local t = i / steps
            setc(math.floor(px + (sx - px) * t + 0.5), math.floor(py + (sy - py) * t + 0.5),
                 s.z < 0.5 and "8" or "7")
          end
          setc(sx, sy, s.z > 0.9 and "7" or s.z > 0.55 and "8" or s.c)
          if s.z < 0.35 then setc(sx + 1, sy, s.c) end
        end
      end
    end,
  }

  -- ПЛАЗМА
  local PL = "fb393ba262ab"
  modes.plasma = {
    init = function() end,
    draw = function(f)
      local t = f * 0.06
      local cx, cy = W / 2, H / 2
      for y = 1, H do
        for x = 1, W do
          local dx, dy = x - cx, (y - cy) * 2
          local v = math.sin(x * 0.17 + t) + math.sin(y * 0.3 - t * 1.3)
                  + math.sin((x + y * 2) * 0.09 + t * 0.7)
                  + math.sin(math.sqrt(dx * dx + dy * dy) * 0.16 - t)
          local i = math.floor((v + 4) / 8 * #PL) + 1
          if i < 1 then i = 1 elseif i > #PL then i = #PL end
          bg[y][x] = PL:sub(i, i)
        end
      end
    end,
  }

  -- СНЕГ
  local flakes
  modes.snow = {
    init = function()
      flakes = {}
      for i = 1, math.max(10, math.floor(W * H / 9)) do
        flakes[i] = {x = math.random(W), y = math.random(H),
                     v = 0.2 + math.random() * 0.5, d = math.random() * 6}
      end
    end,
    draw = function()
      for _, s in ipairs(flakes) do
        s.y = s.y + s.v
        s.x = s.x + math.sin((s.y + s.d) * 0.7) * 0.25
        if s.y > H + 1 then s.y = 0; s.x = math.random(W) end
        local x, y = math.floor(s.x + 0.5), math.floor(s.y)
        if x >= 1 and x <= W and y >= 1 and y <= H then
          txt[y][x] = s.v > 0.5 and "*" or "."
          fg[y][x] = s.v > 0.5 and "0" or "8"
        end
      end
    end,
  }

  -- большие часы + две маленькие строки
  local function drawClock(info)
    local str = string.format("%02d:%02d:%02d", info.h, info.m, info.s)
    local rows = {"", "", "", "", ""}
    for i = 1, #str do
      local ch = str:sub(i, i)
      local g
      if ch == ":" then
        g = (info.s % 2 == 0) and {"0", "1", "0", "1", "0"} or {"0", "0", "0", "0", "0"}
      else
        g = FONT[ch]
      end
      for r = 1, 5 do rows[r] = rows[r] .. g[r] .. (i < #str and "0" or "") end
    end
    local U = #rows[1]
    local pw = math.max(1, math.floor(W / (U + 2)))
    local ph = math.max(1, math.floor(pw * 2 / 3 + 0.5))
    while pw > 1 and 5 * ph + 3 > H do
      pw = pw - 1
      ph = math.max(1, math.floor(pw * 2 / 3 + 0.5))
    end

    if W < U * pw or 5 * ph + 3 > H then
      local mid = math.max(1, math.floor(H / 2) - 1)
      text(mid, str, "0")
      text(mid + 1, info.line1, "8")
      text(mid + 2, info.line2, info.night and "3" or "4")
      return
    end

    local bh = 5 * ph
    local y0 = math.max(1, math.floor((H - (bh + 3)) / 2) + 1)
    local x0 = math.floor((W - U * pw) / 2) + 1
    for pass = 1, 2 do
      for r = 1, 5 do
        for u = 1, U do
          if rows[r]:sub(u, u) == "1" then
            for dy = 0, ph - 1 do
              for dx = 0, pw - 1 do
                local x = x0 + (u - 1) * pw + dx
                local y = y0 + (r - 1) * ph + dy
                if pass == 1 then setc(x + 1, y + 1, "f") else setc(x, y, "0") end
              end
            end
          end
        end
      end
    end
    text(y0 + bh + 1, info.line1, "8")
    text(y0 + bh + 2, info.line2, info.night and "3" or "4")
  end

  function S.setMode(name)
    alloc()
    S.mode = modes[name]
    S.mode.init()
  end

  function S.render(f, info, ui)
    blank()
    S.mode.draw(f)
    drawClock(info)
    if ui then
      text(H - 1, ui.msg, ui.mc)
      text(H, ui.prompt, ui.pc)
    end
    for y = 1, H do
      out.setCursorPos(1, y)
      out.blit(table.concat(txt[y]), table.concat(fg[y]), table.concat(bg[y]))
    end
  end

  alloc()
  return S
end

----------------------------------------------------------------
-- данные для часов
----------------------------------------------------------------
local function zoneName()
  if tz == 3 then return "MSK" end
  return "UTC" .. (tz >= 0 and "+" or "") .. tostring(tz)
end

local function info()
  local secs = math.floor(os.epoch("utc") / 1000 + tz * 3600)
  local d = os.date("!*t", secs)
  local t = os.time()
  local gh = math.floor(t)
  local gm = math.floor((t - gh) * 60)
  return {
    h = d.hour, m = d.min, s = d.sec,
    night = (t >= 18.5 or t < 5.5),
    line1 = string.format("%s %02d.%02d.%d  %s", WDAY[d.wday], d.day, d.month, d.year, zoneName()),
    line2 = string.format("Minecraft %02d:%02d   Day %d", gh, gm, os.day()),
  }
end

-- экраны и смена фонов
----------------------------------------------------------------
math.randomseed(os.epoch("utc"))
local scenes = {makeScene(term.current())}
local mon = peripheral.find("monitor")
if mon then scenes[2] = makeScene(mon) end

local NAMES = {"fire", "matrix", "stars", "plasma", "snow"}
local curMode = NAMES[math.random(#NAMES)]
local nextSwitch = os.epoch("utc") + SWITCH_MS

local function applyMode()
  for _, s in ipairs(scenes) do s.setMode(curMode) end
end

local function switchMode()
  local n
  repeat n = NAMES[math.random(#NAMES)] until n ~= curMode
  curMode = n
  nextSwitch = os.epoch("utc") + SWITCH_MS
  applyMode()
end

local function clearScreens()
  for _, s in ipairs(scenes) do
    s.out.setBackgroundColor(colors.black)
    s.out.setTextColor(colors.white)
    s.out.clear()
    s.out.setCursorPos(1, 1)
  end
end

local function drain()
  local t = os.startTimer(0.3)
  repeat
    local e, a = os.pullEventRaw()
  until e == "timer" and a == t
end

-- ввод пароля; Ctrl+T здесь не работает
local function ask(prompt, msg, mc, lock)
  local buf, frame = "", 0
  local unlock = os.epoch("utc") + (lock or 0) * 1000
  term.setCursorBlink(false)

  local function render()
    local now = os.epoch("utc")
    if now >= nextSwitch then switchMode() end
    local m, c, p = msg or "", mc or "0", prompt .. string.rep("*", #buf) .. "_"
    if now < unlock then
      m = m .. "  wait " .. math.ceil((unlock - now) / 1000) .. " s"
      c = "e"
      p = "Locked"
    end
    local inf = info()
    for i, s in ipairs(scenes) do
      if i == 1 then
        s.render(frame, inf, {msg = m, mc = c, prompt = p, pc = "0"})
      else
        s.render(frame, inf, {msg = "", mc = "0", prompt = "LOCKED", pc = "e"})
      end
    end
    frame = frame + 1
  end

  render()
  local timer = os.startTimer(0.1)
  while true do
    local e, a = os.pullEventRaw()
    if e == "timer" and a == timer then
      render()
      timer = os.startTimer(0.1)
    elseif e == "term_resize" or e == "monitor_resize" then
      applyMode()
    elseif os.epoch("utc") >= unlock then
      if e == "char" then
        if #buf < 40 then buf = buf .. a end
      elseif e == "key" then
        if a == keys.backspace then buf = buf:sub(1, -2)
        elseif a == keys.enter or a == keys.numPadEnter then return buf end
      end
    end
  end
end

local function newPassword(firstMsg)
  local msg, mc = firstMsg, "0"
  while true do
    local p1 = ask("New password: ", msg, mc)
    if #p1 < 4 then
      msg, mc = "At least 4 characters", "e"
    else
      local p2 = ask("Repeat password: ", "", "0")
      if p1 == p2 then saveRec(p1) return end
      msg, mc = "Passwords do not match", "e"
    end
  end
end

-- вход: при первом запуске создаёт пароль
local function lock(firstMsg)
  local rec = loadRec()
  if not rec then
    newPassword(firstMsg or "First start: create a password")
    return
  end
  local fails, msg, mc, wait = 0, "", "0", 0
  while true do
    local p = ask("Password: ", msg, mc, wait)
    if p ~= "" then
      if hash(p, rec.salt) == rec.hash then return end
      fails = fails + 1
      msg, mc, wait = "Wrong password", "e", math.min(30, fails * 3)
    else
      wait = 0
    end
  end
end

local function changePassword()
  local rec = loadRec()
  if rec then
    local msg, fails = "Change password", 0
    while true do
      local p = ask("Current password: ", msg, "0")
      if hash(p, rec.salt) == rec.hash then break end
      fails = fails + 1
      if fails >= 3 then return false end
      msg = "Wrong password"
    end
  end
  newPassword("Choose a new password")
  return true
end

-- скринсейвер: то же самое, но без пароля, закрывается любой клавишей
local function saver()
  applyMode()
  local frame = 0
  local timer = os.startTimer(0.1)
  while true do
    local e, a = os.pullEventRaw()
    if e == "timer" and a == timer then
      if os.epoch("utc") >= nextSwitch then switchMode() end
      local inf = info()
      for _, s in ipairs(scenes) do s.render(frame, inf, nil) end
      frame = frame + 1
      timer = os.startTimer(0.1)
    elseif e == "term_resize" or e == "monitor_resize" then
      applyMode()
    elseif e == "key" or e == "char" or e == "mouse_click"
        or e == "mouse_drag" or e == "monitor_touch" then
      break
    end
  end
  drain()
end

----------------------------------------------------------------
-- запуск
----------------------------------------------------------------
local function main()
  applyMode()
  lock()
  clearScreens()
  drain()

  local history = {}
  if term.isColor() then term.setTextColor(colors.lime) end
  print("Commands: saver, login (lock), passwd, exit")
  term.setTextColor(colors.white)

  while true do
    if term.isColor() then term.setTextColor(colors.yellow) end
    write("/" .. shell.dir() .. "> ")
    term.setTextColor(colors.white)

    local line
    local gotIdle = false
    local idleTimer = os.startTimer(IDLE)

    parallel.waitForAny(
      function()
        line = read(nil, history, shell.complete)
      end,
      function()
        while true do
          local e, a = os.pullEvent()
          if e == "timer" and a == idleTimer then
            gotIdle = true
            return
          elseif e == "key" or e == "char" or e == "paste"
              or e == "mouse_click" or e == "mouse_scroll" then
            idleTimer = os.startTimer(IDLE)
          end
        end
      end
    )

    if gotIdle then
      saver()
      if LOCK_AFTER_SAVER then lock() end
      clearScreens()
      drain()
    else
      print()
      if line and line:match("%S") then
        if history[#history] ~= line then history[#history + 1] = line end
        local cmd = line:match("^%s*(%S+)")
        if cmd == "saver" then
          saver()
          clearScreens()
        elseif cmd == "login" then
          lock()
          clearScreens()
          drain()
        elseif cmd == "passwd" then
          local ok = changePassword()
          clearScreens()
          drain()
          print(ok and "Password changed" or "Password not changed")
        elseif cmd == "exit" then
          break
        else
          shell.run(line)
        end
      end
    end
  end
end

local ok, err = pcall(main)
if not ok then
  clearScreens()
  term.setTextColor(colors.red)
  print("startup error:")
  print(tostring(err))
  term.setTextColor(colors.white)
  print("Press any key...")
  os.pullEventRaw("key")
end
